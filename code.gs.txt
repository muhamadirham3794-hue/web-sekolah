/**
 * KONFIGURASI UTAMA
 */
const SPREADSHEET_ID = '18X_wJhlWagA2uoIVubixS2FGccUScvgoJyJQUwB-aL0';

const SHEET_NAMES = {
  CONFIG: 'Site_Config',
  PPDB: 'DB_PPDB_SMP_ALFATHONAH'
};

/**
 * Helper: Merespons output JSON dengan header CORS lengkap
 */
function createJsonResponse(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

/**
 * Helper: Menangani Preflight CORS Request (OPTIONS)
 */
function doOptions(e) {
  return createJsonResponse({ status: 'success', message: 'CORS Preflight OK' });
}

/**
 * Helper: Mengambil atau Membuat Sheet secara Aman
 */
function getSheet(sheetName) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  let sheet = ss.getSheetByName(sheetName);

  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
    if (sheetName === SHEET_NAMES.CONFIG) {
      sheet.getRange('A1').setValue('{}');
    } else if (sheetName === SHEET_NAMES.PPDB) {
      sheet.appendRow([
        'Waktu Pendaftaran', 
        'No Registrasi', 
        'Nama Lengkap', 
        'NISN',
        'Tempat Lahir', 
        'Tanggal Lahir', 
        'Jenis Kelamin', 
        'Agama',
        'Alamat Lengkap', 
        'Asal Sekolah', 
        'Tahun Lulus', 
        'Nama Orang Tua',
        'Pekerjaan Orang Tua', 
        'No WhatsApp'
      ]);
      // Format header menjadi BOLD
      sheet.getRange('A1:N1').setFontWeight('bold');
    }
  }

  return sheet;
}

/**
 * GET Handler: Membaca Konfigurasi Website (CMS)
 */
function doGet(e) {
  try {
    const sheetConfig = getSheet(SHEET_NAMES.CONFIG);
    const configJsonRaw = sheetConfig.getRange('A1').getValue();
    let configData = {};

    if (configJsonRaw) {
      try {
        configData = JSON.parse(configJsonRaw);
      } catch (err) {
        configData = {};
      }
    }

    return createJsonResponse({
      status: 'success',
      siteData: configData // <--- Sesuaikan dari 'data' menjadi 'siteData'
    });
  } catch (error) {
    return createJsonResponse({
      status: 'error',
      message: error.toString()
    });
  }
}

/**
 * POST Handler: Menyimpan Data (SAVE_CMS & SUBMIT_PPDB)
 */
function doPost(e) {
  const lock = LockService.getScriptLock();
  let hasLock = false;

  try {
    // Kunci thread maksimal 30 detik untuk menghindari data bentrok
    hasLock = lock.tryLock(30000);
    
    if (!hasLock) {
      return createJsonResponse({
        status: 'error',
        message: 'Server sedang sibuk memproses data lain. Silakan coba beberapa saat lagi.'
      });
    }

    if (!e || !e.postData || !e.postData.contents) {
      throw new Error("Payload request kosong.");
    }

    let postData;
    try {
      postData = JSON.parse(e.postData.contents);
    } catch (parseErr) {
      throw new Error("Format JSON tidak valid.");
    }

    // Ekstraksi Action & Payload
    const action = postData.action || (postData.data && postData.data.action);
    const payload = postData.data !== undefined ? postData.data : postData;

    switch (action) {
      case 'SAVE_CMS':
        return handleSaveCms(payload);

      case 'SUBMIT_PPDB':
        return handleSubmitPpdb(payload);

      default:
        throw new Error(`Aksi '${action}' tidak dikenal atau tidak disertakan.`);
    }

  } catch (error) {
    return createJsonResponse({
      status: 'error',
      message: error.message || error.toString()
    });
  } finally {
    if (hasLock) {
      lock.releaseLock();
    }
  }
}

/**
 * Sub-Handler: Menyimpan Data Konfigurasi CMS
 */
function handleSaveCms(payload) {
  const sheetConfig = getSheet(SHEET_NAMES.CONFIG);
  sheetConfig.getRange('A1').setValue(JSON.stringify(payload));

  return createJsonResponse({
    status: 'success',
    message: 'Konfigurasi website berhasil disimpan dan dipublikasikan!'
  });
}

/**
 * Sub-Handler: Menyimpan Form Pendaftaran PPDB
 */
function handleSubmitPpdb(payload) {
  const sheetPPDB = getSheet(SHEET_NAMES.PPDB);
  const d = payload || {};

  // Waktu Pendaftaran (WIB)
  const now = new Date();
  const timestamp = Utilities.formatDate(now, 'Asia/Jakarta', 'yyyy-MM-dd HH:mm:ss');
  
  // Auto-generate No Registrasi jika tidak diset dari frontend (Contoh: REG-20260811-XXXX)
  const autoNoReg = d.noRegistrasi || ('REG-' + Utilities.formatDate(now, 'Asia/Jakarta', 'yyyyMMdd') + '-' + Math.floor(1000 + Math.random() * 9000));

  // Menyiapkan baris data dengan format string aman (mencegah auto-format angka di Sheets)
  const rowData = [
    timestamp,
    String(autoNoReg),
    String(d.namaLengkap || ''),
    "'" + String(d.nisn || ''), // Prefix "'" agar NISN tidak terpotong nol di depan
    String(d.tempatLahir || ''),
    String(d.tanggalLahir || ''),
    String(d.jenisKelamin || ''),
    String(d.agama || ''),
    String(d.alamatLengkap || ''),
    String(d.asalSekolah || ''),
    String(d.tahunLulus || ''),
    String(d.namaOrangTua || ''),
    String(d.pekerjaan || d.pekerjaanOrangTua || ''),
    "'" + String(d.noWhatsapp || '') // Prefix "'" agar No WA seperti 08123... tidak terpotong
  ];

  sheetPPDB.appendRow(rowData);

  return createJsonResponse({
    status: 'success',
    message: 'Pendaftaran PPDB berhasil disimpan!',
    noRegistrasi: autoNoReg
  });
}
